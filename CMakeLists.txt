cmake_minimum_required(VERSION 3.16)
project(cpp_vector_math LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(math3d INTERFACE)
target_include_directories(math3d INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(math_tests tests/math_tests.cpp)
target_link_libraries(math_tests PRIVATE math3d)
enable_testing()
add_test(NAME math_tests COMMAND math_tests)
option(BUILD_DEMO "Build raylib visual demo" ON)
if(BUILD_DEMO)
 find_package(raylib CONFIG REQUIRED)
 add_executable(vector_demo src/main.cpp)
 target_link_libraries(vector_demo PRIVATE math3d raylib)
endif()
